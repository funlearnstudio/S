cmake_minimum_required(VERSION 3.20)
project(S VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(s_core
  src/lexer/token.cpp src/lexer/lexer.cpp src/parser/parser.cpp
  src/runtime/error.cpp src/runtime/value.cpp src/modules/modules.cpp src/ffi/ffi.cpp
  src/interpreter/interpreter.cpp src/type/checker.cpp src/compiler/compiler.cpp)
target_include_directories(s_core PUBLIC include)
target_compile_definitions(s_core PRIVATE S_SOURCE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}" S_NATIVE_BUILD_DIR="${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(s_core PUBLIC ${CMAKE_DL_LIBS})
if(MSVC)
  target_compile_options(s_core PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(s_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

add_executable(s src/main.cpp)
target_link_libraries(s PRIVATE s_core)
target_compile_definitions(s PRIVATE S_SOURCE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}" S_NATIVE_BUILD_DIR="${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
  target_compile_options(s PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(s PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

add_library(native_test SHARED native/native_test.c)
set_target_properties(native_test PROPERTIES OUTPUT_NAME native_test)
if(MSVC)
  target_compile_options(native_test PRIVATE /W4 /WX)
else()
  target_compile_options(native_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(s_tests tests/tests.cpp)
  target_link_libraries(s_tests PRIVATE s_core)
  target_compile_definitions(s_tests PRIVATE S_TEST_ROOT="${CMAKE_CURRENT_SOURCE_DIR}" S_TEST_NATIVE_DIR="${CMAKE_CURRENT_BINARY_DIR}")
  if(MSVC)
    target_compile_options(s_tests PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(s_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME s_tests COMMAND s_tests)
  set_tests_properties(s_tests PROPERTIES ENVIRONMENT "S_NATIVE_PATH=${CMAKE_CURRENT_BINARY_DIR}")
  add_test(NAME cli_check COMMAND s check ${CMAKE_SOURCE_DIR}/examples/hello.s)
  add_test(NAME cli_run COMMAND s run ${CMAKE_SOURCE_DIR}/examples/function.s)
  add_test(NAME cli_type COMMAND s run ${CMAKE_SOURCE_DIR}/examples/type.s)
  add_test(NAME cli_modules COMMAND s run ${CMAKE_SOURCE_DIR}/examples/modules/main.s)
  add_test(NAME cli_collections COMMAND s run ${CMAKE_SOURCE_DIR}/examples/collections.s)
  add_test(NAME cli_errors COMMAND s run ${CMAKE_SOURCE_DIR}/examples/errors.s)
  add_test(NAME cli_native COMMAND s run ${CMAKE_SOURCE_DIR}/examples/native/main.s)
  set_tests_properties(cli_native PROPERTIES ENVIRONMENT "S_NATIVE_PATH=${CMAKE_CURRENT_BINARY_DIR}")
  add_test(NAME parity COMMAND ${CMAKE_COMMAND} -DS_EXE=$<TARGET_FILE:s> -DS_ROOT=${CMAKE_SOURCE_DIR} -DS_WORK=${CMAKE_CURRENT_BINARY_DIR}/parity -DS_NATIVE_DIR=${CMAKE_CURRENT_BINARY_DIR} -P ${CMAKE_SOURCE_DIR}/tests/parity.cmake)
endif()

install(TARGETS s s_core RUNTIME DESTINATION bin ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY std/ DESTINATION share/s/std OPTIONAL)
