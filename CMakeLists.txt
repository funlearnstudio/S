cmake_minimum_required(VERSION 3.20)
project(SE VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(se_core
  src/lexer/token.cpp src/lexer/lexer.cpp src/parser/parser.cpp
  src/runtime/error.cpp src/runtime/value.cpp src/modules/modules.cpp src/ffi/ffi.cpp src/ffi/bindgen.cpp
  src/interpreter/interpreter.cpp src/type/checker.cpp src/compiler/compiler.cpp)
target_include_directories(se_core PUBLIC include)
target_compile_definitions(se_core PRIVATE SE_SOURCE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}" SE_NATIVE_BUILD_DIR="${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(se_core PUBLIC ${CMAKE_DL_LIBS})
if(MSVC)
  target_compile_options(se_core PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(se_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

add_executable(se src/main.cpp)
target_link_libraries(se PRIVATE se_core)
target_compile_definitions(se PRIVATE SE_SOURCE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}" SE_NATIVE_BUILD_DIR="${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
  target_compile_options(se PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(se PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

add_library(native_test SHARED native/native_test.c)
target_include_directories(native_test PRIVATE include native)
set_target_properties(native_test PROPERTIES OUTPUT_NAME native_test)
if(MSVC)
  target_compile_options(native_test PRIVATE /W4 /WX)
else()
  target_compile_options(native_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(se_tests tests/tests.cpp)
  target_link_libraries(se_tests PRIVATE se_core)
  target_compile_definitions(se_tests PRIVATE S_TEST_ROOT="${CMAKE_CURRENT_SOURCE_DIR}" S_TEST_NATIVE_DIR="${CMAKE_CURRENT_BINARY_DIR}")
  if(MSVC)
    target_compile_options(se_tests PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(se_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()

  add_executable(se_ffi_bytes_tests tests/ffi_bytes.cpp)
  target_link_libraries(se_ffi_bytes_tests PRIVATE se_core)
  target_compile_definitions(se_ffi_bytes_tests PRIVATE S_TEST_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
  if(MSVC)
    target_compile_options(se_ffi_bytes_tests PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(se_ffi_bytes_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()

  add_test(NAME se_tests COMMAND se_tests)
  set_tests_properties(se_tests PROPERTIES ENVIRONMENT "SE_NATIVE_PATH=${CMAKE_CURRENT_BINARY_DIR}")
  add_test(NAME ffi_bytes COMMAND se_ffi_bytes_tests)
  set_tests_properties(ffi_bytes PROPERTIES ENVIRONMENT "SE_NATIVE_PATH=${CMAKE_CURRENT_BINARY_DIR}")
  add_test(NAME cli_check COMMAND se check ${CMAKE_SOURCE_DIR}/examples/hello.se)
  add_test(NAME cli_run COMMAND se run ${CMAKE_SOURCE_DIR}/examples/function.se)
  add_test(NAME cli_type COMMAND se run ${CMAKE_SOURCE_DIR}/examples/type.se)
  add_test(NAME cli_modules COMMAND se run ${CMAKE_SOURCE_DIR}/examples/modules/main.se)
  add_test(NAME cli_collections COMMAND se run ${CMAKE_SOURCE_DIR}/examples/collections.se)
  add_test(NAME cli_errors COMMAND se run ${CMAKE_SOURCE_DIR}/examples/errors.se)
  add_test(NAME cli_native COMMAND se run ${CMAKE_SOURCE_DIR}/examples/native/main.se)
  set_tests_properties(cli_native PROPERTIES ENVIRONMENT "SE_NATIVE_PATH=${CMAKE_CURRENT_BINARY_DIR}")
  add_test(NAME parity COMMAND ${CMAKE_COMMAND} -DS_EXE=$<TARGET_FILE:se> -DS_ROOT=${CMAKE_SOURCE_DIR} -DS_WORK=${CMAKE_CURRENT_BINARY_DIR}/parity -DS_NATIVE_DIR=${CMAKE_CURRENT_BINARY_DIR} -P ${CMAKE_SOURCE_DIR}/tests/parity.cmake)
  if(NOT WIN32)
    add_test(NAME bindgen COMMAND ${CMAKE_COMMAND} -DS_EXE=$<TARGET_FILE:se> -DS_ROOT=${CMAKE_SOURCE_DIR} -DS_WORK=${CMAKE_CURRENT_BINARY_DIR}/bindgen -DS_CC=${CMAKE_C_COMPILER} -DS_CXX=${CMAKE_CXX_COMPILER} -P ${CMAKE_SOURCE_DIR}/tests/bindgen.cmake)
  endif()
endif()

install(TARGETS se RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION share/se/include)
install(DIRECTORY std/ DESTINATION share/se/std OPTIONAL)
install(FILES src/runtime/error.cpp src/runtime/value.cpp DESTINATION share/se/src/runtime)
install(FILES src/interpreter/interpreter.cpp DESTINATION share/se/src/interpreter)
install(FILES src/ffi/ffi.cpp DESTINATION share/se/src/ffi)

set(CPACK_PACKAGE_NAME "SE")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_VENDOR "funlearnstudio")
set(CPACK_PACKAGE_FILE_NAME "se-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}")
if(WIN32)
  set(CPACK_GENERATOR "ZIP")
else()
  set(CPACK_GENERATOR "TGZ")
endif()
include(CPack)
